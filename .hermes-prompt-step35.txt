export type FieldAccess = 'hidden' | 'readonly' | 'readwrite';

export interface TablePermission {
  canRead: boolean;
  canCreate: boolean;
  canEdit: boolean;
  canDelete: boolean;
}

export interface FieldPermission {
  fieldName: string;
  access: FieldAccess;
}

export interface UserPermissions {
  roleIds: number[];
  roleNames: string[];
  isAdmin: boolean;
  tablePerms: Record<string, TablePermission>;
  fieldPerms: Record<string, Record<string, FieldAccess>>; // tableName -> { fieldName -> access }
  rowFilters: Record<string, string>;
}

export type PermissionAction = 'read' | 'create' | 'edit' | 'delete';
```

### 3. Create src/permissions/usePermissions.ts

interface UsePermissionsReturn {
  /** Is the user an admin? (bypasses all checks) */
  isAdmin: boolean;

  can: (action: PermissionAction) => boolean;
  /** Loading state */
  loading: boolean;
  /** Error state */
  error: string | null;
  /** Refetch permissions */

```

The hook should:
- Accept `tableName: string` as argument
- Use usePermissionsContext() to get the current user's permissions
- Look up the table in the permissions map
- For getFieldAccess: check field_permissions for the specific table+field, return 'readwrite' if no specific permission exists
- If isAdmin, return true for all actions and 'readwrite' for all fields
- Use React cache/SWR-like pattern: fetch permissions once on mount, cache in React context

### 4. Create src/permissions/PermissionGate.tsx

A component that conditionally renders children based on permission checks:

```typescript
interface PermissionGateProps {
  /** Table to check permissions for */
  table: string;
  /** Required action (default: 'read') */
  action?: PermissionAction;
  /** Field name to check (optional — checks field-level access) */
  field?: string;
  /** Required field access level (default: 'readwrite' — only applies if field is set) */
  fieldAccess?: FieldAccess;
  /** Fallback content when not permitted (default: null) */
  fallback?: React.ReactNode;
  /** Children to render when permitted */
  children: React.ReactNode;
}
```

Behavior:
- Renders children if the user has the required permission
- Renders `fallback` (or null) if the user doesn't have the permission
- If `field` is provided, checks field-level access instead of table-level action
- The fallback for field='hidden' should render nothing by default
- If loading, renders nothing (or a spinner if loadingFallback is provided)
- Add an optional `loadingFallback?: React.ReactNode` prop

### 5. Create src/permissions/FieldPermissionGate.tsx

A simpler component focused on field-level rendering in forms:

```typescript
interface FieldPermissionGateProps {
  table: string;
  field: string;
  /** What to render when the field is hidden (default: <></>) */
  hiddenFallback?: React.ReactNode;
  /** Children to render when access is 'readwrite' or 'readonly' */
  children: (props: { readOnly: boolean; hidden: boolean }) => React.ReactNode;
}
```

This is a render-prop style component that passes readOnly and hidden flags to its children, allowing form renderers to adjust field rendering without knowing about permissions.
